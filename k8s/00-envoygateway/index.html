    Https#

    一张 SAN 证书，同时包含 whoami.devops.io 和 www.devops.io
    • 生成 SAN 证书（同时带两个域名）
    cat > openssl.cnf <<EOF
[req]
distinguished_name = dn
x509_extensions = v3_ext
prompt = no

[dn]
CN = devops.io

[v3_ext]
subjectAltName = @alt_names
basicConstraints = CA:FALSE
keyUsage = digitalSignature,keyEncipherment

[alt_names]
DNS.1 = whoami.devops.io
DNS.2 = www.devops.io
EOF
openssl req -x509 -newkey rsa:4096 -keyout tls.key -out tls.crt -days 3650 -nodes -config openssl.cnf
    • 创建 secret（envoy-gateway-system）
    kubectl create secret tls devops-tls-secret --cert=tls.crt --key=tls.key -n envoy-gateway-system

    # 1. 增加https listener
kubectl apply -f - << EOF
apiVersion: gateway.networking.k8s.io/v1

    两个独立证书，两个 HTTPS listener（各自域名独立证书）
    • 生成证书和secret
    # 证书1：whoami.devops.io
openssl req -x509 -newkey rsa:4096 -keyout whoami.key -out whoami.crt -days 3650 -nodes -subj "/CN=whoami.devops.io"
# 证书2：www.devops.io
openssl req -x509 -newkey rsa:4096 -keyout www.key -out www.crt -days 3650 -nodes -subj "/CN=www.devops.io"
kubectl create secret tls whoami-tls-secret --cert=whoami.crt --key=whoami.key -n envoy-gateway-system
kubectl create secret tls www-tls-secret --cert=www.crt --key=www.key -n envoy-gateway-system

    # 两个 HTTPS listener，都在 443 端口，靠 `hostname` 区分域名（SNI）
kubectl apply -f - << EOF
apiVersion: gateway.networking.k8s.io/v1
kind: Gateway
metadata:
  name: eg
  namespace: envoy-gateway-system
spec:
  gatewayClassName: eg
  listeners:
  - name: http
    protocol: HTTP
    port: 80
    allowedRoutes:
      namespaces:
        from: All

  - name: https-whoami
    protocol: HTTPS
    port: 443
    hostname: whoami.devops.io
    tls:
      mode: Terminate
      certificateRefs:
      - kind: Secret
        name: whoami-tls-secret
        namespace: envoy-gateway-system
    allowedRoutes:
      namespaces:
        from: All

  - name: https-www
    protocol: HTTPS
    port: 443
    hostname: www.devops.io
    tls:
      mode: Terminate
      certificateRefs:
      - kind: Secret
        name: www-tls-secret
        namespace: envoy-gateway-system
    allowedRoutes:
      namespaces:
        from: All
EOF

    kubectl apply -f - <<EOF
apiVersion: gateway.networking.k8s.io/v1
kind: HTTPRoute
metadata:
  name: www
  namespace: default
spec:
  parentRefs:
  - name: eg
    namespace: envoy-gateway-system
    sectionName: https-www
  hostnames:
  - "www.devops.io"
  rules:
  - matches:
    - path:
        type: PathPrefix
        value: /
    backendRefs:
    - name: whoami
      port: 80
EOF

    kubectl apply -f - << EOF
apiVersion: gateway.networking.k8s.io/v1
kind: HTTPRoute
metadata:
  name: redirect-https
  namespace: default
spec:
  parentRefs:
  - name: eg
    namespace: envoy-gateway-system
    sectionName: http
  hostnames:
  - whoami.devops.io
  - www.devops.io
  rules:
  - matches:
    - path:
        type: PathPrefix
        value: /
    filters:
    - type: RequestRedirect
      requestRedirect:
        scheme: https
        statusCode: 302
EOF
